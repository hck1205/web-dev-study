```js
// 회의는 [시작, 끝). 10시에 끝나는 회의와 10시에 시작하는 회의는 겹치지 않는다
function pickGeneral(meetings, key) {
  const sorted = [...meetings].sort(key);
  const chosen = [];
  for (const m of sorted)
    if (chosen.every((c) => m[1] <= c[0] || m[0] >= c[1])) chosen.push(m); // 이미 고른 것과 안 겹치면 넣는다
  return chosen.sort((a, b) => a[0] - b[0]);
}
const byShort = (a, b) => (a[1] - a[0]) - (b[1] - b[0]); // 짧은 회의부터
const byStart = (a, b) => a[0] - b[0];                     // 일찍 시작하는 회의부터
const byEnd = (a, b) => a[1] - b[1];                       // 일찍 끝나는 회의부터

const A = [[9, 12], [11, 13], [12, 15]];
const B = [[9, 18], [10, 11], [11, 12], [13, 14]];
for (const [name, ms] of [["A", A], ["B", B]])
  for (const [kname, k] of [["short", byShort], ["start", byStart], ["end", byEnd]]) {
    const r = pickGeneral(ms, k);
    console.log(name, kname.padEnd(5), r.length, JSON.stringify(r));
  }
```

매 순간 정해진 기준으로 "지금 가장 좋아 보이는 것"을 고르고, 한 번 고른 것은 다시 돌아보지 않는 방법을 그리디(greedy, 탐욕) 알고리즘이라고 한다. 빠르고 코드가 짧다. 대신 위처럼 기준을 잘못 고르면 그럴듯하게 틀린다. 이 글은 어떤 기준이 왜 맞는지 증명하는 법, 틀리는 기준을 빨리 찾아내는 법, 그리디가 실패하는 대표 문제와 그때 넘어갈 곳을 다룬다. 출력은 모두 Node.js 22에서 실제로 돌린 결과다.

넘어갈 곳은 대개 [동적 계획법](/posts/dynamic-programming)이다. 그리디는 정렬 하나로 끝나는 경우가 많아 [정렬 알고리즘](/posts/sorting-algorithms)과도 붙어 다닌다. 복잡도를 세는 법은 [Big O 표기법](/posts/big-o-notation)에 있다.

```js
function brute(ms) {                      // 모든 부분집합 중 서로 안 겹치는 가장 큰 것
  let best = 0;
  for (let mask = 0; mask < 1 << ms.length; mask++) {
    const s = ms.filter((_, i) => (mask >> i) & 1);
    const ok = s.every((x, i) => s.every((y, j) => i === j || x[1] <= y[0] || y[1] <= x[0]));
    if (ok) best = Math.max(best, s.length);
  }
  return best;
}
// mulberry32로 시드를 고정한 난수. 시작 0~19시, 길이 1~8시간인 회의 6개를 2,000번 만든다
for (const [kname, k] of [["short", byShort], ["start", byStart], ["end", byEnd]]) {
  let fails = 0, first = null;
  for (let t = 0; t < 2000; t++) {
    const ms = Array.from({ length: 6 }, () => { const s = rnd(20); return [s, s + 1 + rnd(8)]; });
    const g = pickGeneral(ms, k).length, b = brute(ms);
    if (g !== b) { fails++; if (!first) first = { ms, g, b }; }
  }
  console.log(`${kname}: ${fails}/2000 wrong`, first ? JSON.stringify(first) : "");
}
```

## 일찍 끝나는 회의가 왜 맞는가

스트레스 테스트가 0번 틀렸다고 맞는 건 아니다. 면접에서 "왜 맞느냐"를 물으면 쓰는 증명 방법이 교환 논증(exchange argument)이다. 생각의 줄기는 이렇다.

어떤 최적해 O가 있다고 하자. 그리디가 처음 고른 회의 g는 모든 회의 중 가장 일찍 끝난다. O의 첫 회의 o1은 g보다 늦게 끝나거나 같이 끝난다. 이제 O에서 o1을 빼고 g를 넣는다. g는 o1보다 일찍 끝나니 o1 뒤에 있던 O의 두 번째 회의와도 겹치지 않는다. 회의 수는 그대로다. 그러니 "g로 시작하는 최적해"가 반드시 존재한다. g를 고른 뒤 남은 문제는 g가 끝난 다음에 시작하는 회의들로 같은 질문을 하는 것이고, 같은 논리를 반복하면 그리디가 끝까지 최적해와 개수가 같다.

이 증명에서 두 성질이 드러난다. 하나는 탐욕 선택 속성이다. 지금의 탐욕스러운 선택을 포함하는 최적해가 항상 있다. 다른 하나는 최적 부분 구조다. 그 선택을 하고 남은 문제의 최적해를 붙이면 전체 최적해가 된다. 최적 부분 구조는 [동적 계획법](/posts/dynamic-programming)에도 필요한 성질이다. 그리디는 여기에 "선택지를 다 따져 볼 필요 없이 하나로 정해진다"는 조건이 더 붙는 셈이다. 그래서 DP보다 적용 범위가 좁고 대신 훨씬 빠르다.

짧은 순이 왜 틀리는지도 같은 틀로 설명된다. 짧은 회의를 고르면 그 회의가 양옆을 막아서, 최적해의 회의 두 개를 회의 하나로 바꾸는 일이 생긴다. 교환하면 개수가 줄어드니 교환 논증이 성립하지 않는다.

코드는 정렬 한 번과 훑기 한 번이다. 매번 이미 고른 것 전체와 비교할 필요 없이 마지막으로 고른 회의의 끝 시각 하나만 기억하면 된다.

```js
function maxMeetings(meetings) {
  const sorted = [...meetings].sort((a, b) => a[1] - b[1]); // 끝나는 시각 순
  let count = 0, lastEnd = -Infinity;
  for (const [s, e] of sorted) if (s >= lastEnd) { count++; lastEnd = e; }
  return count;
}
```

정렬이 O(n log n), 훑기가 O(n)이라 전체 O(n log n)이다. `s >= lastEnd`의 등호가 "10시에 끝나고 10시에 시작하면 안 겹친다"는 규칙이다. 구간이 양 끝을 포함하는 문제라면 `>`로 바꿔야 하고, 면접에서 이 경계를 먼저 물어보면 좋은 인상을 준다.

## 회의실이 몇 개 필요한가

조건을 바꾸면 다른 질문이 된다. 모든 회의를 다 열어야 할 때 회의실이 최소 몇 개 필요한가(LeetCode 253 Meeting Rooms II). 여기서도 그리디가 통한다. 시각 순으로 훑으면서 "지금 동시에 열려 있는 회의 수"의 최댓값을 구하면 된다. 시작 시각과 끝 시각을 따로 정렬해 두 포인터로 훑는다.

```js
function minRooms(ms) {
  const starts = ms.map((m) => m[0]).sort((a, b) => a - b);
  const ends = ms.map((m) => m[1]).sort((a, b) => a - b);
  let rooms = 0, best = 0, j = 0;
  for (let i = 0; i < starts.length; i++) {
    while (j < ends.length && ends[j] <= starts[i]) { j++; rooms--; } // 이미 끝난 회의는 방을 비운다
    rooms++;                                                         // 새 회의가 방 하나를 쓴다
    best = Math.max(best, rooms);
  }
  return best;
}
console.log("minRooms", minRooms([[9, 10], [9, 12], [10, 11], [11, 13], [12, 13]]));
```

구간 문제는 이 밖에도 자주 나온다. 겹치는 구간을 하나로 합치기(56번)는 시작 순으로 정렬하고 마지막 구간의 끝을 늘려 가는 그리디다. 겹치지 않게 지워야 하는 최소 구간 수(435번)는 "최대로 남길 수 있는 회의 수"의 여집합이라 끝 순 정렬 그대로다. 정렬 기준이 시작이냐 끝이냐가 문제마다 다르고, 그걸 고르는 근거가 교환 논증이다.

## 큰 동전부터 내면 틀리는 나라

그리디의 반례로 가장 유명한 게 거스름돈이다. 큰 단위부터 최대한 내는 방법은 한국 동전(10, 50, 100, 500원)에서는 언제나 최소 개수다. 그런데 단위를 조금만 바꾸면 틀린다. 금액 2,000까지 그리디와 DP의 답을 전부 비교해 처음 어긋나는 금액을 찾았다.

```js
function greedyCount(coins, a) {
  let n = 0;
  for (const c of [...coins].sort((x, y) => y - x)) { n += Math.floor(a / c); a %= c; }
  return a === 0 ? n : Infinity;
}
// dpAll: 동적 계획법으로 0~limit 모든 금액의 최소 개수를 구한다 (동적 계획법 글의 bottomUp과 같다)
function firstCounter(coins, limit) {
  const dp = dpAll(coins, limit);
  for (let x = 1; x <= limit; x++)
    if (greedyCount(coins, x) !== dp[x]) return { amount: x, greedy: greedyCount(coins, x), best: dp[x] };
  return null;
}
for (const coins of [[1, 3, 4], [10, 50, 100, 500], [1, 5, 10, 25], [1, 10, 25], [1, 7, 10]])
  console.log(JSON.stringify(coins), "first counterexample:", JSON.stringify(firstCounter(coins, 2000)));
```

```text

[10,50,100,500] first counterexample: null
[1,5,10,25] first counterexample: null
[1,10,25] first counterexample: {"amount":30,"greedy":6,"best":3}
[1,7,10] first counterexample: {"amount":14,"greedy":5,"best":2}
```

미국 동전(1, 5, 10, 25센트)에서 5센트를 빼면 30센트가 25 + 1×5로 6개, 정답은 10×3의 3개다. [1, 7, 10]에서는 14가 10 + 1×4로 5개, 정답은 7×2다. 그리디가 언제나 맞는 단위 체계를 정규(canonical) 동전 체계라고 하고, 실제 화폐는 대부분 이렇게 설계돼 있다. 정규가 아닌 체계라면 가장 작은 반례가 가장 큰 두 단위의 합보다 작은 범위에 있다는 결과(Kozen과 Zaks, 1994)가 있어서 위처럼 그 범위까지만 비교해 보면 판정된다.

이게 실무에서 의미가 있는 건 쿠폰, 포인트, 묶음 상품처럼 단위를 회사가 정하는 경우다. 상품을 1개, 3개 묶음, 4개 묶음으로 판다면 "큰 묶음부터 담기"는 손님에게 최선이 아니다. 단위를 새로 정할 때 위 함수 같은 검사를 한 번 돌려 보면 된다.

## 쪼갤 수 있으면 그리디, 못 쪼개면 DP

배낭 문제도 같은 경계를 보여 준다. 물건을 잘라서 일부만 담을 수 있으면(분할 가능 배낭) 무게당 가치가 높은 것부터 담는 그리디가 최적이다. 자를 수 없으면(0/1 배낭) 같은 그리디가 틀린다. 물건 A(무게 10, 가치 60), B(20, 100), C(30, 120), 용량 50으로 셋을 비교했다.

```js
const items = [{ name: "A", w: 10, v: 60 }, { name: "B", w: 20, v: 100 }, { name: "C", w: 30, v: 120 }];
const byRatio = [...items].sort((a, b) => b.v / b.w - a.v / a.w); // 무게당 가치: A 6, B 5, C 4
function fractional(W) {
  let left = W, total = 0; const log = [];
  for (const it of byRatio) {
    const take = Math.min(it.w, left);       // 남은 만큼만 잘라 담는다
    total += (take * it.v) / it.w; left -= take; log.push(`${it.name} ${take}/${it.w}`);
    if (!left) break;
  }
  return { total, log };
}
function greedy01(W) {
  let left = W, total = 0; const got = [];
  for (const it of byRatio) if (it.w <= left) { left -= it.w; total += it.v; got.push(it.name); }
  return { total, got };
}
console.log("fractional", JSON.stringify(fractional(50)));
console.log("greedy 0/1", JSON.stringify(greedy01(50)));
console.log("dp 0/1", dp01(items, 50)); // 동적 계획법 글의 1차원 배낭
```

```text
fractional {"total":240,"log":["A 10/10","B 20/20","C 20/30"]}
greedy 0/1 {"total":160,"got":["A","B"]}
dp 0/1 220
```

자를 수 있으면 C를 3분의 2만 담아 240이다. 자를 수 없으면 비율 순 그리디는 A, B를 담고 남은 20칸에 C가 안 들어가 160에서 멈춘다. 정답은 B + C의 220이다. 교환 논증으로 보면 이유가 보인다. 분할 가능할 때는 비율이 낮은 물건의 일부를 비율이 높은 물건의 같은 무게로 바꾸면 가치가 늘거나 같으니 교환이 성립한다. 자를 수 없으면 무게가 딱 맞지 않아 그 교환을 할 수 없다. 남는 칸이 생기는 순간 그리디의 선택은 되돌릴 수 없는 손해가 된다.

## 멀리 갈 수 있는 곳만 기억하기

그리디가 정렬 없이 한 번 훑기로 끝나는 경우도 있다. 점프 게임(55번)은 배열의 각 칸에 그 칸에서 최대 몇 칸까지 뛸 수 있는지 적혀 있을 때 마지막 칸에 닿을 수 있는지 묻는다. 모든 점프 경로를 따지면 DP나 백트래킹이 되지만, "지금까지 닿을 수 있는 가장 먼 칸" 하나만 기억하면 충분하다.

```js
function canJump(nums) {
  let reach = 0;
  for (let i = 0; i < nums.length; i++) {
    if (i > reach) return false;              // 여기까지 오지도 못한다
    reach = Math.max(reach, i + nums[i]);     // 이 칸에서 더 멀리 갈 수 있으면 늘린다
  }
  return true;
}
console.log("canJump", canJump([2, 3, 1, 1, 4]), canJump([3, 2, 1, 0, 4]));
```

## 자주 나오는 글자에 짧은 부호를

압축에서 쓰는 허프만 코딩은 그리디로 최적이 증명된 대표 예다. 글자마다 길이가 다른 비트 부호를 붙이되, 자주 나오는 글자일수록 짧게 붙여 전체 비트 수를 최소로 만든다. 방법은 단순하다. 빈도가 가장 작은 두 덩어리를 꺼내 하나로 합치기를 하나 남을 때까지 반복한다. 합친 순서가 그대로 트리가 되고, 왼쪽을 0, 오른쪽을 1로 읽으면 부호가 나온다.

"abracadabra"로 돌려 봤다. 가장 작은 둘을 꺼내는 부분은 설명을 위해 매번 정렬했다. 실제로는 [우선순위 큐](/posts/ds-heap)를 써서 O(k log k)로 만든다(k는 글자 종류 수).

```js
function huffman(text) {
  const freq = new Map();
  for (const ch of text) freq.set(ch, (freq.get(ch) ?? 0) + 1);
  let nodes = [...freq].map(([ch, f]) => ({ ch, f }));
  const steps = [];
  while (nodes.length > 1) {
    nodes.sort((a, b) => a.f - b.f || String(a.ch ?? "~").localeCompare(String(b.ch ?? "~")));
    const [x, y] = nodes.splice(0, 2);                       // 가장 드문 두 덩어리
    steps.push(`${x.ch ?? "(" + x.f + ")"}+${y.ch ?? "(" + y.f + ")"}=${x.f + y.f}`);
    nodes.push({ f: x.f + y.f, l: x, r: y });                // 합쳐서 다시 넣는다
  }
  const codes = {};
  (function walk(n, p) {
    if (n.ch !== undefined) { codes[n.ch] = p || "0"; return; }
    walk(n.l, p + "0"); walk(n.r, p + "1");
  })(nodes[0], "");
  const bits = [...text].reduce((s, ch) => s + codes[ch].length, 0);
  return { freq: Object.fromEntries(freq), codes, bits, fixed: text.length * Math.ceil(Math.log2(freq.size)), steps };
}
console.log(JSON.stringify(huffman("abracadabra")));
```


## 그래프에서 그리디가 맞을 때와 틀릴 때

그래프 알고리즘 중에도 그리디가 많다. 최소 신장 트리의 크루스칼은 "가장 싼 간선부터, 고리를 만들지 않으면 고른다"는 그리디이고, 고리 검사는 [유니온 파인드](/posts/ds-union-find)가 맡는다. 프림도 "지금 트리에 붙일 수 있는 가장 싼 간선"을 고르는 그리디다. 최단 경로의 다익스트라는 "아직 확정 안 된 정점 중 가장 가까운 것"을 확정하고 다시 보지 않는다.

다익스트라의 그리디는 간선 무게가 음수가 아니라는 가정 위에 서 있다. 지금 가장 가까운 정점은 다른 길로 돌아와도 더 가까워질 수 없다는 가정이다. 음수 간선 하나로 그 가정을 깨 봤다.

```js
const edges = [["S", "A", 2], ["S", "B", 5], ["B", "A", -4], ["A", "T", 3]];
const nodes = ["S", "A", "B", "T"];
function dijkstra(src) {               // 설명용으로 힙 없이 가장 가까운 정점을 매번 찾는다
  const dist = Object.fromEntries(nodes.map((n) => [n, Infinity])); dist[src] = 0;
  const done = new Set(); const order = [];
  while (done.size < nodes.length) {
    let u = null;
    for (const n of nodes) if (!done.has(n) && (u === null || dist[n] < dist[u])) u = n;
    done.add(u); order.push(`${u}=${dist[u]}`);           // 확정하고 다시 보지 않는다
    for (const [a, b, w] of edges) if (a === u && !done.has(b) && dist[u] + w < dist[b]) dist[b] = dist[u] + w;
  }
  return { dist, order };
}
function bellman(src) {                // 모든 간선을 V-1번 다시 본다
  const dist = Object.fromEntries(nodes.map((n) => [n, Infinity])); dist[src] = 0;
  for (let i = 0; i < nodes.length - 1; i++)
    for (const [a, b, w] of edges) if (dist[a] + w < dist[b]) dist[b] = dist[a] + w;
  return dist;
}
console.log("dijkstra", JSON.stringify(dijkstra("S")));
console.log("bellman ", JSON.stringify(bellman("S")));
```

## 현장에서 그리디를 만나는 곳

그리디는 최적이 증명된 곳에서만 쓰는 게 아니다. 최적을 구하기 너무 비싼 문제에서 "충분히 좋은 답을 빨리" 내는 근사 알고리즘으로 훨씬 많이 쓰인다.

쿠버네티스 스케줄러가 대표적이다. kube-scheduler는 파드를 한 번에 하나씩 꺼내 조건에 맞는 노드를 거르고(filter), 남은 노드에 점수를 매겨(score) 가장 높은 노드에 놓는다. 앞으로 올 파드를 함께 고려해 전체 배치를 최적화하지 않는다. 그래서 작은 파드들이 노드마다 조금씩 자리를 차지해 큰 파드가 들어갈 노드가 없어지는 조각화가 생긴다. 이미 놓인 파드를 다시 옮겨 일부 완화하는 도구로 descheduler가 있다. 증상은 "클러스터 전체로는 CPU가 남는데 파드가 Pending"이고, `kubectl describe pod`의 FailedScheduling 이벤트에 `Insufficient cpu`가 노드 수와 함께 찍힌다.

상자 채우기(bin packing)도 같다. 크기가 다른 짐을 최소 개수의 상자에 담는 문제는 NP-난해라 정확한 답을 구하기 어렵다. 큰 짐부터 정렬해 들어가는 첫 상자에 넣는 First Fit Decreasing은 최적 개수의 11/9배에 작은 상수를 더한 값을 넘지 않는다는 보장이 있어서 실무의 배치, 메모리 할당, 클라우드 인스턴스 채우기에서 기본값처럼 쓰인다. [로드 밸런싱 알고리즘](/posts/load-balancing-algorithms)의 least connections도 "지금 가장 한가한 서버"를 고르는 그리디이고, 그 순간의 판단이 나중에 틀릴 수 있다는 한계를 그대로 가진다.

이런 곳에서 그리디를 쓸 때는 "얼마나 나쁠 수 있나"를 알아 두는 게 중요하다. 최악의 입력에서 최적의 몇 배까지 나빠지는지(근사 비율)가 알려진 방법을 고르고, 그 비율이 서비스에 허용되는지 따진다.

## 그리디인지 확인하는 순서

그리디 풀이가 떠올랐을 때 이 순서로 검증하면 틀린 풀이를 제출할 일이 줄어든다.

- 기준을 한 문장으로 쓴다. "끝나는 시각이 이른 것부터", "무게당 가치가 높은 것부터"처럼.
- 손으로 반례를 찾아본다. 짧은 것, 긴 것, 동점, 하나가 여럿을 막는 배치를 넣어 본다.
- 작은 입력으로 완전 탐색과 비교하는 스트레스 테스트를 돌린다. 몇천 번에 반례가 안 나오면 다음 단계로 간다.
- 교환 논증을 시도한다. 최적해의 첫 선택을 그리디의 선택으로 바꿔도 손해가 없는지 본다.
- 교환이 안 되면 DP로 넘어간다. 선택이 뒤의 선택지를 바꾸고 되돌릴 수 없는 손해가 생기는 문제다.

## 면접에서 나올 만한 질문

### 그리디 알고리즘은 무엇이고 언제 최적해를 보장하나요?

매 단계 정해진 기준으로 지금 가장 좋아 보이는 선택을 하고 되돌아보지 않는 방법이다. 지금의 탐욕 선택을 포함하는 최적해가 항상 있고(탐욕 선택 속성), 그 선택 뒤에 남은 문제의 최적해를 붙이면 전체 최적해가 될 때(최적 부분 구조) 최적을 보장한다. 회의 배정의 "끝나는 시각이 이른 순"이 대표적이다.

### 회의실 배정에서 짧은 회의부터 고르면 왜 틀리나요?

짧은 회의 하나가 양옆의 회의 두 개와 겹쳐 둘을 동시에 막을 수 있기 때문이다. [9,12], [11,13], [12,15]에서 짧은 순은 [11,13] 하나만 넣지만 정답은 양 끝 둘이다. 시드를 고정한 무작위 입력 2,000개로 완전 탐색과 비교하니 짧은 순은 67번, 일찍 시작하는 순은 397번 틀렸고 끝나는 순은 한 번도 안 틀렸다.

### 교환 논증으로 끝나는 시각 기준이 맞다는 걸 설명해 보세요.

아무 최적해를 하나 잡고, 그 첫 회의를 그리디가 고른 가장 일찍 끝나는 회의로 바꾼다. 그리디의 회의는 최적해의 첫 회의보다 늦게 끝나지 않으니 최적해의 두 번째 회의와도 겹치지 않고 개수는 그대로다. 그러면 그리디의 첫 선택을 포함하는 최적해가 존재하고, 남은 문제에 같은 논리를 반복하면 그리디가 최적해와 개수가 같다.

### 거스름돈 문제에서 그리디는 항상 최소 개수를 주나요?

아니다. 단위 체계에 따라 다르다. 한국 동전이나 미국 동전(1, 5, 10, 25)은 그리디가 항상 맞는 정규 체계지만, [1, 3, 4]로 6을 만들면 그리디는 4+1+1의 3개, 정답은 3+3의 2개다. [1, 10, 25]에서도 30이 반례다. 단위를 회사가 정하는 쿠폰이나 묶음 상품이라면 DP와 비교해 반례가 있는지 확인한다.

### 분할 가능 배낭은 그리디로 풀리는데 0/1 배낭은 왜 안 되나요?

분할 가능하면 비율이 낮은 물건의 일부를 비율이 높은 물건의 같은 무게로 바꾸는 교환이 항상 가능해서 비율 순 그리디가 최적이다. 0/1에서는 물건을 자를 수 없어 무게가 딱 맞는 교환이 안 되고 남는 칸이 손해로 남는다. 무게 10/20/30, 가치 60/100/120, 용량 50에서 비율 순 그리디는 160, 정답은 220이었다.

### 다익스트라는 왜 음수 간선에서 틀리나요?

가장 가까운 정점을 확정하면 다른 길로 돌아와도 더 가까워질 수 없다는 그리디 가정을 쓰기 때문이다. 음수 간선이 있으면 나중에 꺼낸 정점을 거쳐 이미 확정한 정점이 더 가까워질 수 있다. 예제에서 다익스트라는 T까지 5, 벨만-포드는 4를 냈다. 음수 간선이 있으면 모든 간선을 V-1번 다시 보는 벨만-포드를 쓴다.

### 그리디 풀이가 맞는지 확신이 없을 때 어떻게 하나요?

작은 입력을 무작위로 많이 만들어 완전 탐색 결과와 비교하는 스트레스 테스트부터 돌린다. 틀린 기준은 보통 몇천 번 안에 반례를 내놓고, 그 반례를 보면 왜 틀렸는지가 보인다. 반례가 없으면 교환 논증으로 증명을 시도하고, 교환이 성립하지 않으면 DP로 넘어간다.

### 최적이 아닌 그리디를 실무에서 쓰는 이유는 무엇인가요?

정확한 최적해가 NP-난해이거나 너무 비싼 문제에서 충분히 좋은 답을 빠르게 내기 때문이다. 쿠버네티스 스케줄러는 파드를 하나씩 점수가 가장 높은 노드에 놓는 그리디라서 조각화가 생길 수 있고, 상자 채우기의 First Fit Decreasing은 최적의 약 11/9배를 넘지 않는다. 이런 곳에서는 최악에 얼마나 나빠질 수 있는지 근사 비율을 알고 쓴다.
