```js
const items = Array.from({ length: 45 }, (_, i) => `상품 ${i + 1}`);
const PAGE_SIZE = 20;

function pageOf(search) {
  const page = parseInt(new URLSearchParams(search).get("page"));
  return items.slice((page - 1) * PAGE_SIZE, page * PAGE_SIZE);
}

for (const q of ["?page=2", "?page=two", "", "?page=-1"]) {
  console.log(JSON.stringify(q), pageOf(q).length, pageOf(q).slice(0, 2));
}
```

필요한 건 "값이 없거나 틀렸다"는 사실이 계산을 따라 흘러가다가 끝에서 한 번에 처리되는 구조다. 이 글은 그 구조를 `map`과 `flatMap`이라는 익숙한 메서드에서 출발해 만든다. 함수형 프로그래밍에서 펑터(functor), 모나드(monad)라고 부르는 것들인데, 이름은 마지막에 붙인다. 출력은 Node.js 22와 TypeScript 5.9에서 실제로 돌린 결과다.

이 주제의 다른 글: 함수형 프로그래밍의 전체 그림은 [함수형 프로그래밍 기초](/posts/functional-programming-basics)에, 함수를 이어 붙이는 `pipe`는 [compose와 pipe](/posts/fp-composition-pipe)에 있다. 배열의 `flatMap` 자체는 [배열 변환 메서드](/posts/js-array-transform)에서, Promise의 `then` 규칙은 [Promise](/posts/js-promises)에서 다뤘다.

## 상자 안의 값을 꺼내지 않고 바꾼다

숫자를 두 배로 만드는 함수 하나를 세 가지 상자에 넣어 본다.

```js
const double = (n) => n * 2;

console.log([1, 2, 3].map(double));                    // 배열: 값이 여러 개
console.log(await Promise.resolve(21).then(double));   // Promise: 값이 나중에 온다
const user = { coupon: { rate: 0.1 } };
console.log(user.coupon?.rate, ({}).coupon?.rate);     // ?.: 값이 없을 수 있다
```

```text
[ 2, 4, 6 ]
42
0.1 undefined
```

세 줄 모두 `double`은 숫자 하나만 안다. 배열인지, 아직 오지 않은 값인지, 없을 수도 있는 값인지는 상자가 처리한다. 배열은 원소마다 부르고, Promise는 값이 도착하면 부르고, `?.`는 값이 있을 때만 뒤를 평가한다. 함수를 상자 안으로 들여보내고, 결과를 같은 종류의 상자에 다시 담아 준다.

이렇게 "안의 값에 함수를 적용하고 같은 모양의 상자를 돌려주는 `map`"이 있는 것을 펑터라고 부른다. 정식으로는 두 가지 약속을 지켜야 한다. 아무것도 안 하는 함수로 `map`하면 원래와 같아야 하고, 두 번 `map`한 결과와 두 함수를 합친 함수로 한 번 `map`한 결과가 같아야 한다.

```js
const id = (x) => x;
const inc = (n) => n + 1;
const xs = [1, 2, 3];
console.log(JSON.stringify(xs.map(id)) === JSON.stringify(xs));
console.log(JSON.stringify(xs.map(inc).map(double)), JSON.stringify(xs.map((x) => double(inc(x)))));
```

```text
true
[4,6,8] [4,6,8]
```

약속이 실무에서 뜻하는 건 하나다. `map`은 값만 바꾸고 상자의 모양(원소 개수, 성공 여부, 순서)은 건드리지 않는다. 그래서 `.map(f).map(g)`를 `.map(x => g(f(x)))`로 합쳐도 되고, 반대로 쪼개도 된다. `map` 안에서 원소를 빼거나 순서를 바꾸고 싶다면 그건 `map`이 아니라 `filter`나 `sort`가 할 일이다.

`?.`는 상자처럼 동작하지만 `null`과 `undefined`에만 반응한다. `parseInt("two")?.toFixed()`는 `"NaN"`을 돌려준다. `NaN`은 값이 있는 것으로 보기 때문이다. 처음 버그를 `?.`로 막을 수 없는 이유다. "무엇을 실패로 볼지"를 우리가 정하는 상자가 필요하다.

## 상자가 두 겹이 될 때

`map`에 넘긴 함수가 상자를 돌려주면 상자 안에 상자가 생긴다.

```js
const words = (s) => s.split(" ");
const lines = ["a b", "c"];
console.log(lines.map(words), lines.flatMap(words));
```

```text
[ [ 'a', 'b' ], [ 'c' ] ] [ 'a', 'b', 'c' ]
```

`words`는 문자열 하나를 받아 배열(상자)을 돌려준다. `map`은 그 결과를 또 배열에 담아 두 겹이 되고, `flatMap`은 한 겹을 벗겨 이어 붙인다. "상자를 돌려주는 함수"를 이어 붙일 때 필요한 게 `flatMap`이다. 파이프의 단계가 실패할 수 있는 함수라면 단계마다 상자를 돌려주니, 이어 붙이려면 `flatMap`이 있어야 한다.

Promise에는 `flatMap`이 따로 없다. `then` 하나가 두 일을 다 한다. 핸들러가 값을 돌려주면 `map`처럼, Promise를 돌려주면 `flatMap`처럼 한 겹을 벗긴다.

```js
const getUser = (id) => Promise.resolve({ id, teamId: 7 });
const getTeam = (u) => Promise.resolve({ team: "payments", of: u.id });
console.log(await Promise.resolve(1).then(getUser).then(getTeam));
console.log(await Promise.resolve(1).then((x) => Promise.resolve(Promise.resolve(x))));
```

```text
{ team: 'payments', of: 1 }
1
```

두 겹, 세 겹으로 감싸도 한 번의 `await`에 값이 나온다. 자바스크립트에서는 "Promise를 담은 Promise"를 만들 수 없다. 편하지만, 이 자동 펼치기는 `then` 메서드가 있는 모든 객체(thenable)에 적용된다는 함정이 있다.

```js
function queryBuilder(table) {
  const q = {
    table, wheres: [],
    where(c) { q.wheres.push(c); return q; },
    then(resolve) {
      console.log(`실행: SELECT * FROM ${q.table} WHERE ${q.wheres.join(" AND ")}`);
      resolve([{ id: 1 }]);
    },
  };
  return q;
}

async function buildUserQuery() {
  return queryBuilder("users").where("active = true"); // 나중에 조건을 더 붙이려고 빌더를 돌려준다
}
const result = await buildUserQuery();
console.log("받은 것:", result);
```

```text
실행: SELECT * FROM users WHERE active = true
받은 것: [ { id: 1 } ]
```

빌더를 돌려받아 조건을 더 붙이려 했는데, `async` 함수가 빌더의 `then`을 불러서 쿼리가 그 자리에서 실행됐고 결과 배열이 왔다. knex 같은 쿼리 빌더는 실제로 `then`을 가지고 있어서 이런 일이 생긴다. 아래 그림에서 다음을 누르면 그 과정이 보인다.

## 없는 값을 상자로 만든다

`flatMap`이 생겼으니 "없을 수 있는 값" 상자를 직접 만들 수 있다. 함수형 언어에서 Maybe나 Option이라고 부르는 것이다. 값이 있으면 `Some`, 없으면 `None`이고, `flatMap`은 `None`을 만나면 뒤의 함수를 부르지 않고 `None`을 그대로 넘긴다.

```js
const Some = (value) => ({ some: true, value });
const None = { some: false };
const flatMapM = (m, f) => (m.some ? f(m.value) : m);
const getOr = (m, fallback) => (m.some ? m.value : fallback);

const param = (search, key) => {
  const v = new URLSearchParams(search).get(key);
  return v === null ? None : Some(v);
};
const toInt = (s) => (/^\d+$/.test(s) ? Some(Number(s)) : None);
const positive = (n) => (n >= 1 ? Some(n) : None);

const pageFrom = (search) => getOr(flatMapM(flatMapM(param(search, "page"), toInt), positive), 1);
for (const q of ["?page=2", "?page=two", "", "?page=-1", "?page=0"]) console.log(JSON.stringify(q), "→", pageFrom(q));
```

```text
"?page=2" → 2
"?page=two" → 1
"" → 1
"?page=-1" → 1
"?page=0" → 1
```

`NaN`이나 음수가 계산에 들어갈 길이 없어졌다. `toInt`와 `positive`가 "무엇이 실패인가"를 정하고, 실패는 끝의 `getOr`에서 한 번에 기본값 1이 된다.

## 실패에 이유를 붙이면 Result가 된다

Maybe는 왜 없는지 말해 주지 않는다. `?page=two`(숫자가 아님)와 `?page=9`(마지막 페이지 너머)는 사용자에게 다르게 안내해야 할 수도 있다. 실패 쪽에도 값을 싣는 상자가 Result(다른 이름으로 Either)다.

```js
const ok = (value) => ({ ok: true, value });
const err = (error) => ({ ok: false, error });
const map = (r, f) => (r.ok ? ok(f(r.value)) : r);
const flatMap = (r, f) => (r.ok ? f(r.value) : r);

const parsePage = (search) => {
  const raw = new URLSearchParams(search).get("page");
  if (raw === null) return ok(1);
  if (!/^\d+$/.test(raw)) return err({ code: "NOT_A_NUMBER", raw });
  return ok(Number(raw));
};
const inRange = (max) => (n) => (n >= 1 && n <= max ? ok(n) : err({ code: "OUT_OF_RANGE", n, max }));

const lastPage = Math.ceil(45 / 20);
for (const q of ["?page=2", "?page=two", "", "?page=9"]) {
  const r = map(flatMap(parsePage(q), inRange(lastPage)), (n) => ({ from: (n - 1) * 20, to: n * 20 }));
  console.log(JSON.stringify(q), JSON.stringify(r));
}
```

```text
"?page=2" {"ok":true,"value":{"from":20,"to":40}}
"?page=two" {"ok":false,"error":{"code":"NOT_A_NUMBER","raw":"two"}}
"" {"ok":true,"value":{"from":0,"to":20}}
"?page=9" {"ok":false,"error":{"code":"OUT_OF_RANGE","n":9,"max":3}}
```

이 그림이 모나드라는 이름의 실체다. 상자를 만드는 방법(`ok`)과 상자를 돌려주는 함수를 이어 붙이는 방법(`flatMap`)이 있고, 이어 붙이는 순서를 어떻게 묶든 결과가 같다는 약속을 지키는 것. Array의 `flatMap`, 직접 만든 Maybe와 Result가 그렇다. Promise의 `then`은 거의 그렇지만 thenable을 무조건 펼쳐서 "Promise를 담은 Promise"를 만들 수 없으니 엄밀한 의미의 모나드는 아니다. 약속 세 가지를 Result로 확인해 봤다.

```js
const half = (n) => (n % 2 === 0 ? ok(n / 2) : err(`${n}은 홀수`));
const minus1 = (n) => (n > 0 ? ok(n - 1) : err("0 이하"));
const same = (a, b) => JSON.stringify(a) === JSON.stringify(b);
for (const x of [8, 3]) {
  console.log(x,
    same(flatMap(ok(x), half), half(x)),     // ok로 감싸 바로 넘기면 함수를 부른 것과 같다
    same(flatMap(half(x), ok), half(x)),     // ok로 다시 감싸기만 하면 그대로다
    same(flatMap(flatMap(ok(x), half), minus1), flatMap(ok(x), (v) => flatMap(half(v), minus1)))); // 묶는 순서가 상관없다
}
```

```text
8 true true true
3 true true true
```

세 번째 약속 덕분에 긴 체인을 중간에서 잘라 이름 붙은 함수로 빼도 결과가 같다. 리팩터링할 때 믿을 수 있는 성질이다.

## 예외와 무엇이 다른가

같은 일을 예외로도 할 수 있다. `parsePage`가 숫자가 아니면 던지고, 바깥에서 `try/catch`로 받으면 된다. 둘의 차이는 실패가 타입과 코드에 보이느냐다.

```ts
type Result<T, E> = { ok: true; value: T } | { ok: false; error: E };
type PageError = { code: "NOT_A_NUMBER"; raw: string } | { code: "OUT_OF_RANGE"; n: number; max: number };

declare function parsePageOrThrow(search: string): number;
declare function parsePage(search: string): Result<number, PageError>;

const a: number = parsePageOrThrow("?page=two"); // 던질 수 있다는 사실이 타입에 없다

const r = parsePage("?page=two");
const b: number = r.value; // 확인 없이 꺼내면

if (r.ok) {
  const c: number = r.value;
} else {
  switch (r.error.code) {
    case "NOT_A_NUMBER": console.log(r.error.raw); break;
    case "OUT_OF_RANGE": console.log(r.error.max); break;
  }
}
```

이 파일을 `tsc --strict`로 검사하면 에러가 하나 난다(경로는 줄였다).

```text
…/result.ts(10,21): error TS2339: Property 'value' does not exist on type 'Result<number, PageError>'.
  Property 'value' does not exist on type '{ ok: false; error: PageError; }'.
```

`parsePageOrThrow`를 부르는 7번째 줄은 아무 말 없이 통과한다. TypeScript에는 함수가 무엇을 던지는지 적는 문법이 없고, `catch (e)`의 `e`는 `unknown`이다. 반대로 Result는 `ok`를 확인하기 전에는 `value`를 꺼낼 수 없게 막고, `else` 쪽에서는 `error.code`마다 다른 필드(`raw`, `max`)를 정확히 좁혀 준다. 실패를 처리하지 않으면 컴파일이 안 되는 것, 그게 Result를 쓰는 가장 큰 이유다.

그렇다고 예외를 모두 Result로 바꾸자는 건 아니다. 나누는 기준이 있다. 사용자가 틀린 값을 넣었거나 외부 서비스가 404를 준 것처럼 예상할 수 있고 처리 방법이 정해진 실패는 Result로 돌려준다. `undefined`에서 속성을 읽거나 불변식이 깨진 것처럼 코드가 틀린 경우(버그)는 던져서 빨리 죽게 둔다. 버그까지 Result로 감싸면 스택 트레이스 없이 조용히 실패 선로를 달려 처음 `NaN` 버그와 같은 꼴이 된다. Result의 `error`에 평범한 객체만 실으면 스택이 없다는 점도 기억해 둔다. 원인을 추적해야 하는 실패라면 `Error` 객체를 `cause`로 함께 싣는다.

이 모양은 이미 여러 곳에서 쓰인다. Zod의 `safeParse`는 `{ success: true, data }` 또는 `{ success: false, error }`를 돌려준다. 표준의 `Promise.allSettled`도 결과마다 성공과 실패를 값으로 담는다.

```js
const fakeFetch = async (url) => { if (url.includes("404")) throw new Error("HTTP 404"); return { id: 1 }; };
const settled = await Promise.allSettled([fakeFetch("/users/1"), fakeFetch("/users/404")]);
console.log(settled.map((s) => (s.status === "fulfilled" ? s : { status: s.status, reason: s.reason.message })));
```

```text
[
  { status: 'fulfilled', value: { id: 1 } },
  { status: 'rejected', reason: 'HTTP 404' }
]
```

## 비동기 체인에 Result 싣기

Promise와 Result는 함께 쓰는 경우가 많다. 네트워크 요청은 던지는 API(`fetch`, axios)이니, 경계에서 한 번 Result로 바꾸고 안쪽은 Result로 잇는다.

```js
const fakeFetch = async (url) => { // /users/404에서 던지는 가짜 fetch
  if (url.includes("/users/404")) throw new Error("HTTP 404");
  if (url.includes("/teams/")) return { team: "payments" };
  return { id: 1, teamId: 7 };
};
const fromPromise = (p, toError) => p.then(ok, (e) => err(toError(e)));

const getUser = (id) => fromPromise(fakeFetch(`/users/${id}`), (e) => ({ code: "USER_FETCH", message: e.message }));
const getTeam = (u) => fromPromise(fakeFetch(`/teams/${u.teamId}`), (e) => ({ code: "TEAM_FETCH", message: e.message }));

// 비동기 flatMap: 앞이 실패면 뒤를 부르지 않는다
const flatMapAsync = async (rp, f) => { const r = await rp; return r.ok ? f(r.value) : r; };

console.log(JSON.stringify(await flatMapAsync(getUser(1), getTeam)));
console.log(JSON.stringify(await flatMapAsync(getUser(404), getTeam)));
```

```text
{"ok":true,"value":{"team":"payments"}}
{"ok":false,"error":{"code":"USER_FETCH","message":"HTTP 404"}}
```

두 번째 줄에서 `getTeam`은 아예 불리지 않았고, 실패 이유에는 어느 단계였는지(`USER_FETCH`)가 남았다.

라이브러리를 쓰면 이 보일러플레이트가 줄어든다. neverthrow는 `ResultAsync`로 Promise와 Result를 한 상자에 묶고, fp-ts와 Effect는 훨씬 큰 체계를 준다. 다만 API가 버전마다 바뀌어 왔으니 쓰기 전에 현재 문서를 확인한다. 작은 프로젝트라면 위의 함수 네 개(`ok`, `err`, `map`, `flatMap`)와 TypeScript 유니온만으로 충분하다.

## 첫 실패에서 멈추지 않아야 할 때

`flatMap`은 첫 실패에서 멈춘다. 페이지 번호처럼 앞 단계가 성공해야 다음 단계가 의미 있는 경우에는 그게 맞다. 그런데 가입 폼은 다르다. 이메일이 틀렸다고 알려 주고 고치면 그제야 비밀번호도 틀렸다고 알려 주면 사용자는 세 번 제출해야 한다.

```js
const rules = {
  email: (v) => (/^[^@\s]+@[^@\s]+$/.test(v) ? ok(v) : err("이메일 형식이 아니다")),
  password: (v) => (v.length >= 8 ? ok(v) : err("비밀번호는 8자 이상")),
  age: (v) => (Number.isInteger(v) && v >= 14 ? ok(v) : err("14세 이상만 가입")),
};
const firstError = (input) => {
  for (const [k, rule] of Object.entries(rules)) {
    const r = rule(input[k]);
    if (!r.ok) return err({ [k]: r.error });
  }
  return ok(input);
};
const allErrors = (input) => {
  const errors = Object.fromEntries(
    Object.entries(rules).map(([k, rule]) => [k, rule(input[k])]).filter(([, r]) => !r.ok).map(([k, r]) => [k, r.error]),
  );
  return Object.keys(errors).length === 0 ? ok(input) : err(errors);
};
const bad = { email: "kim", password: "1234", age: 12 };
console.log(JSON.stringify(firstError(bad)));
console.log(JSON.stringify(allErrors(bad)));
```

```text
{"ok":false,"error":{"email":"이메일 형식이 아니다","password":"비밀번호는 8자 이상","age":"14세 이상만 가입"}}
```

## Result가 짐이 되는 순간

모든 함수가 Result를 돌려주기 시작하면 비용이 보인다. 던지는 라이브러리와 만날 때마다 감싸야 하고, 감싸는 걸 잊은 곳에서는 예외와 Result가 섞여 두 가지를 다 처리해야 한다. 팀원 모두가 `flatMap`으로 읽고 쓰는 데 익숙하지 않으면 리뷰가 느려진다. 실패할 일이 없는 함수까지 `ok()`로 감싸면 소음만 늘어난다.

그래서 쓰는 자리를 정한다. 사용자 입력을 파싱하고 검증하는 곳, 외부 응답을 도메인 값으로 바꾸는 경계, "재고 부족", "쿠폰 만료"처럼 업무 규칙상 일어나는 실패. 이런 곳에서 Result를 쓰고, 그 안쪽의 순수한 계산은 그냥 값을 돌려주고, 버그는 던진다.

React에서도 같은 생각이 쓰인다. `isLoading`, `isError`, `data`를 따로 들고 있으면 `isLoading`이 `true`인데 `data`도 있는 이상한 조합이 생긴다. `{ status: "pending" } | { status: "error", error } | { status: "success", data }`처럼 하나의 유니온으로 두면 불가능한 조합이 타입에서 사라진다. TanStack Query가 `status`를 이 세 값으로 주는 것도 같은 모양이다. 서버 데이터 캐시 쪽 이야기는 [클라이언트 데이터 캐시](/posts/client-data-cache)에, 컴포넌트 상태를 이렇게 설계하는 법은 [React 안의 함수형](/posts/fp-in-react)에 있다.

## 연습: Result의 map과 flatMap

`map`은 성공이면 함수를 적용해 다시 `ok`로 감싸고, 실패면 그대로 돌려준다. `flatMap`은 성공이면 함수의 결과(이미 Result)를 그대로 돌려주고, 실패면 그대로 돌려준다. `signupAge`는 테스트용이라 고치지 않는다.

## 다시 빈 페이지로

처음 버그는 "숫자가 아니다"라는 사실을 `NaN`이라는 숫자로 표현해서 생겼다. 실패를 상자로 표현하면 계산은 상자 안의 값에만 함수를 적용하고(`map`), 실패할 수 있는 단계는 상자를 돌려주며 이어지고(`flatMap`), 실패는 끝에서 한 번에 처리된다. 배열, Promise, Maybe, Result가 모두 이 두 메서드를 가진 같은 모양이고, 펑터와 모나드는 그 모양의 이름이다. 예상한 실패는 Result로, 버그는 예외로 나눠 두면 둘 다 제 역할을 한다.

## 면접에서 나올 만한 질문

### 펑터란 무엇인가요?

안의 값에 함수를 적용하고 같은 종류의 상자를 돌려주는 `map`을 가진 것이다. 배열의 `map`, Promise의 `then`(값을 돌려주는 경우)이 대표적이다. 아무것도 안 하는 함수로 `map`하면 그대로이고, 두 번 `map`한 것과 합친 함수로 한 번 `map`한 것이 같아야 한다. 그래서 `map`은 값만 바꾸고 원소 개수나 성공 여부 같은 상자의 모양은 바꾸지 않는다.

### 모나드를 쉽게 설명해 주세요.

상자를 돌려주는 함수를 이어 붙일 수 있게 해 주는 `flatMap`을 가진 상자다. `map`으로 이으면 상자 안에 상자가 생기는데, `flatMap`은 한 겹을 벗겨 다음 함수로 넘긴다. 배열의 `flatMap`, Maybe와 Result가 그렇다. 묶는 순서와 상관없이 결과가 같다는 약속 덕분에 긴 체인을 잘라 함수로 빼도 동작이 같다.

### Promise는 모나드인가요?

비슷하지만 엄밀히는 아니다. `then`이 `map`과 `flatMap`을 겸해서, 핸들러가 Promise를 돌려주면 자동으로 펼친다. 그래서 Promise를 담은 Promise를 만들 수 없고, `then` 메서드가 있는 아무 객체나 펼쳐 버린다. `async` 함수에서 쿼리 빌더 같은 thenable을 돌려주면 그 `then`이 불려 쿼리가 실행되는 이유다.

### Result 타입을 쓰면 예외와 비교해 무엇이 좋은가요?

실패가 함수의 타입에 드러나고, 처리하지 않으면 컴파일이 안 된다. TypeScript는 무엇을 던지는지 적을 수 없고 `catch`의 에러는 `unknown`이지만, Result 유니온은 `ok`를 확인하기 전에 값을 못 꺼내게 하고 에러 코드별로 필드를 좁혀 준다. 대신 던지는 라이브러리와 만날 때마다 감싸야 하고 스택 정보가 없으니, 원인이 필요한 실패는 `Error`를 `cause`로 싣는다.

### 예외와 Result는 어떻게 나눠 쓰나요?

예상할 수 있고 처리 방법이 정해진 실패는 Result, 코드가 틀린 버그는 예외다. 사용자 입력 검증, 외부 응답 파싱, 재고 부족 같은 업무 규칙 실패는 값으로 돌려주고 화면에서 안내한다. `undefined` 접근이나 불변식 위반은 던져서 빨리 드러나게 한다. 버그까지 Result로 감싸면 스택 없이 조용히 실패가 흘러가 찾기 어려워진다.

### 폼 검증에서 첫 에러만 보여 주는 게 왜 문제이고 어떻게 바꾸나요?

`flatMap`처럼 첫 실패에서 멈추면 사용자가 하나 고칠 때마다 다음 에러를 보게 되어 여러 번 제출해야 한다. 서로 기대지 않는 필드는 전부 검사해 실패를 모아 한 번에 돌려준다. 앞 결과가 있어야 의미 있는 검사(비밀번호 확인 같은)만 이어서 검사한다. 함수형 라이브러리는 이 모으는 방식을 Validation으로 따로 둔다.

### 비동기 코드에서 Result는 어떻게 쓰나요?

던지는 API를 경계에서 한 번 Result로 바꾼다. `p.then(ok, (e) => err(...))`처럼 거부를 값으로 바꾸고, 안쪽에서는 `await` 뒤에 `ok`를 확인해 다음 단계를 부르는 비동기 `flatMap`으로 잇는다. 실패 이유에 단계 이름을 코드로 붙이면 어디서 실패했는지 남는다. neverthrow의 `ResultAsync` 같은 라이브러리가 이 패턴을 묶어 준다.
